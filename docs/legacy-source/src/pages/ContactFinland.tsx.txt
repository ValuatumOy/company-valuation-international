import { Helmet } from "react-helmet-async";
import PageLayout from "@/components/layout/PageLayout";
import { useState } from "react";
import { useToast } from "@/hooks/use-toast";
import { z } from "zod";

const contactSchema = z.object({
  name: z.string().trim().min(1, "Name is required").max(100, "Name must be less than 100 characters"),
  email: z.string().trim().email("Invalid email address").max(255, "Email must be less than 255 characters"),
  company: z.string().trim().max(200, "Company name must be less than 200 characters").optional(),
  finnishCompany: z.string().trim().min(1, "Please specify the Finnish company for valuation").max(200, "Company name must be less than 200 characters"),
  message: z.string().trim().max(2000, "Message must be less than 2000 characters").optional(),
});

type ContactFormData = z.infer<typeof contactSchema>;

const breadcrumbJsonLd = {
  "@context": "https://schema.org",
  "@type": "BreadcrumbList",
  itemListElement: [
    { "@type": "ListItem", position: 1, name: "Home", item: "https://company-valuation.com/" },
    { "@type": "ListItem", position: 2, name: "Contact: Finland", item: "https://company-valuation.com/contact-finland" },
  ],
};

const ContactFinland = () => {
  const { toast } = useToast();
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [formData, setFormData] = useState<ContactFormData>({
    name: "",
    email: "",
    company: "",
    finnishCompany: "",
    message: "",
  });
  const [errors, setErrors] = useState<Partial<Record<keyof ContactFormData, string>>>({});

  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear error when user starts typing
    if (errors[name as keyof ContactFormData]) {
      setErrors(prev => ({ ...prev, [name]: undefined }));
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    
    // Validate form
    const result = contactSchema.safeParse(formData);
    if (!result.success) {
      const fieldErrors: Partial<Record<keyof ContactFormData, string>> = {};
      result.error.errors.forEach(err => {
        if (err.path[0]) {
          fieldErrors[err.path[0] as keyof ContactFormData] = err.message;
        }
      });
      setErrors(fieldErrors);
      return;
    }

    setIsSubmitting(true);

    // Create mailto link
    const subject = encodeURIComponent(`Credit Risk & Valuation Report Request - ${result.data.finnishCompany}`);
    const body = encodeURIComponent(
      `Name: ${result.data.name}\n` +
      `Email: ${result.data.email}\n` +
      `Company: ${result.data.company || "Not specified"}\n` +
      `Finnish Company for Valuation: ${result.data.finnishCompany}\n` +
      `\nMessage:\n${result.data.message || "No additional message"}`
    );
    
    // Open email client
    window.location.href = `mailto:company-valuation@valuatum.com?subject=${subject}&body=${body}`;

    toast({
      title: "Opening email client",
      description: "Your email client should open with the pre-filled message. If it doesn't, please send an email directly to company-valuation@valuatum.com",
    });

    setIsSubmitting(false);
  };

  return (
    <PageLayout>
      <Helmet>
        <title>Contact: Finnish Company Valuation | Valuatum Tools</title>
        <meta name="description" content="Request a Finnish company valuation report or ask questions about the Finland workflow. We'll reply by email." />
        <script type="application/ld+json">{JSON.stringify(breadcrumbJsonLd)}</script>
      </Helmet>

      <article className="mt-8 mb-7">
        <div className="hero-box-inner">
          <h1 className="page-title">Credit Risk and Valuation Reports - Finland</h1>
          <p className="text-base mb-3 max-w-3xl mx-auto">
            Request a credit risk and valuation report for any Finnish company. Fill out the form below and we will send you the report for a fixed fee.
          </p>
        </div>
      </article>

      <hr className="section-divider" />

      <div className="max-w-2xl mx-auto">
        <section className="block mb-9">
          <div className="text-box">
            <h2 className="block-heading">Request a Report</h2>
            
            <form onSubmit={handleSubmit} className="space-y-6 mt-6">
              <div>
                <label htmlFor="name" className="block text-sm font-medium mb-2">
                  Your Name <span className="text-destructive">*</span>
                </label>
                <input
                  type="text"
                  id="name"
                  name="name"
                  value={formData.name}
                  onChange={handleChange}
                  className={`w-full px-4 py-3 rounded-lg border bg-background focus:outline-none focus:ring-2 focus:ring-primary ${
                    errors.name ? "border-destructive" : "border-border"
                  }`}
                  placeholder="John Doe"
                />
                {errors.name && <p className="text-destructive text-sm mt-1">{errors.name}</p>}
              </div>

              <div>
                <label htmlFor="email" className="block text-sm font-medium mb-2">
                  Your Email <span className="text-destructive">*</span>
                </label>
                <input
                  type="email"
                  id="email"
                  name="email"
                  value={formData.email}
                  onChange={handleChange}
                  className={`w-full px-4 py-3 rounded-lg border bg-background focus:outline-none focus:ring-2 focus:ring-primary ${
                    errors.email ? "border-destructive" : "border-border"
                  }`}
                  placeholder="john@company.com"
                />
                {errors.email && <p className="text-destructive text-sm mt-1">{errors.email}</p>}
              </div>

              <div>
                <label htmlFor="company" className="block text-sm font-medium mb-2">
                  Your Company
                </label>
                <input
                  type="text"
                  id="company"
                  name="company"
                  value={formData.company}
                  onChange={handleChange}
                  className={`w-full px-4 py-3 rounded-lg border bg-background focus:outline-none focus:ring-2 focus:ring-primary ${
                    errors.company ? "border-destructive" : "border-border"
                  }`}
                  placeholder="Your Company Name"
                />
                {errors.company && <p className="text-destructive text-sm mt-1">{errors.company}</p>}
              </div>

              <div>
                <label htmlFor="finnishCompany" className="block text-sm font-medium mb-2">
                  Finnish Company for Valuation <span className="text-destructive">*</span>
                </label>
                <input
                  type="text"
                  id="finnishCompany"
                  name="finnishCompany"
                  value={formData.finnishCompany}
                  onChange={handleChange}
                  className={`w-full px-4 py-3 rounded-lg border bg-background focus:outline-none focus:ring-2 focus:ring-primary ${
                    errors.finnishCompany ? "border-destructive" : "border-border"
                  }`}
                  placeholder="Name of the Finnish company you want a report for"
                />
                {errors.finnishCompany && <p className="text-destructive text-sm mt-1">{errors.finnishCompany}</p>}
              </div>

              <div>
                <label htmlFor="message" className="block text-sm font-medium mb-2">
                  Additional Message
                </label>
                <textarea
                  id="message"
                  name="message"
                  value={formData.message}
                  onChange={handleChange}
                  rows={4}
                  className={`w-full px-4 py-3 rounded-lg border bg-background focus:outline-none focus:ring-2 focus:ring-primary resize-y ${
                    errors.message ? "border-destructive" : "border-border"
                  }`}
                  placeholder="Any additional information or specific requirements..."
                />
                {errors.message && <p className="text-destructive text-sm mt-1">{errors.message}</p>}
              </div>

              <div className="text-center pt-4">
                <button
                  type="submit"
                  disabled={isSubmitting}
                  className="btn btn-primary px-8 py-3 text-base disabled:opacity-50"
                >
                  {isSubmitting ? "Sending..." : "Send Request"}
                </button>
              </div>
            </form>

            <div className="mt-8 p-4 bg-muted rounded-lg">
              <p className="text-sm text-muted-foreground text-center">
                You can also contact us directly at{" "}
                <a href="mailto:company-valuation@valuatum.com" className="text-primary hover:underline">
                  company-valuation@valuatum.com
                </a>
              </p>
            </div>
          </div>
        </section>
      </div>
    </PageLayout>
  );
};

export default ContactFinland;
