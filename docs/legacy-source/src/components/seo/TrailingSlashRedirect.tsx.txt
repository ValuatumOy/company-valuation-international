import { useEffect } from "react";
import { useLocation, useNavigate } from "react-router-dom";

/**
 * Component that redirects URLs with trailing slashes to their non-trailing slash versions.
 * This ensures URL consistency and prevents duplicate content issues for SEO.
 */
const TrailingSlashRedirect = () => {
  const location = useLocation();
  const navigate = useNavigate();

  useEffect(() => {
    // Check if the path has a trailing slash (but is not just "/")
    if (location.pathname.length > 1 && location.pathname.endsWith("/")) {
      // Remove the trailing slash and navigate
      const newPath = location.pathname.slice(0, -1) + location.search + location.hash;
      navigate(newPath, { replace: true });
    }
  }, [location, navigate]);

  return null;
};

export default TrailingSlashRedirect;
